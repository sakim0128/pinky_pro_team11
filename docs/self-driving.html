def _on_command(self, msg: FleetCommand):
    now = self._seconds()
    event = self._link.on_command(
        now, heartbeat=(msg.command == FleetCommand.CMD_HEARTBEAT))
    if event == ARMED:   ...   # 데드맨 감시 시작
    elif event == RESTORED: self._link_lost = False

    if msg.command == FleetCommand.CMD_HEARTBEAT:
        return                       # 수신 시각 갱신이 전부다

    # 끊겼다 붙는 순간 RELIABLE QoS 가 옛 CMD_GOTO 를 재전송한다
    if msg.command in (FleetCommand.CMD_GOTO, FleetCommand.CMD_SET_INITIAL_POSE):
        if not self._link.accepts_motion_command(now):
            return

    if msg.command == FleetCommand.CMD_GOTO:
        self._hold = False
        self._goal = (msg.x, msg.y, msg.yaw); self._goal_valid = True
        self._send_goal()
    elif msg.command == FleetCommand.CMD_STOP:
        self._hold = True; self._hold_since = self.get_clock().now()
        self._cancel_goal()
    elif msg.command == FleetCommand.CMD_RESUME:
        if self._hold and self._goal_valid:
            self._hold = False; self._send_goal()

mini_project_1 — coordinator_node: 교착 감지 → 순차 발진

평소 회피는 각 로봇의 Nav2 로컬 플래너가 한다(서로를 라이다 장애물로 봄). coordinator 는 Nav2 가 스스로 못 빠져나가는 상황만 잡는다.

NORMAL
  둘 다 NAV_ACTIVE + 둘 다 |v| < stall_speed(0.03)
  + 거리 < conflict_distance(0.45)  가 stall_duration(3 s) 지속
      → domain_id 큰 쪽에 CMD_STOP → YIELD

YIELD  (아래 중 하나면 CMD_RESUME → NORMAL, cooldown 2 s)
  · 거리 > clear_distance(0.70)        # 히스테리시스
  · leader 주행 종료 (SUCCEEDED/ABORTED/CANCELED)
  · resume_timeout(15 s) 초과          # 안전장치
시간 상수 순서를 테스트가 지킨다: cooldown(2) < stall_duration(3) < resume_timeout(15) < hold_watchdog(20). 값은 pinky_fleet_station/config/mission.yaml.
pinky_fleet_station/coordinator_node.py (발췌)
def _tick_normal(self, a, b, distance, dt, now):
    if now < self._cooldown_until:
        self._stall_timer = 0.0; return
    both_active = (a.nav_status == RobotState.NAV_ACTIVE
                   and b.nav_status == RobotState.NAV_ACTIVE)
    both_stalled = (abs(a.linear_velocity) < self._stall_speed
                    and abs(b.linear_velocity) < self._stall_speed)
    close = distance < self._conflict_distance
    if a.localized and b.localized and both_active and both_stalled and close:
        self._stall_timer += dt
    else:
        self._stall_timer = 0.0
    if self._stall_timer >= self._stall_duration:
        # domain_id 가 작은 쪽이 leader, 큰 쪽이 yielder
        self._leader, self._yielder = self._pair[0]['name'], self._pair[1]['name']
        self._publish_command(self._yielder, FleetCommand.CMD_STOP)
        self._state = STATE_YIELD
        self._yield_start = now

def _tick_yield(self, a, b, distance, now):
    waited = now - (self._yield_start or now)
    if distance > self._clear_distance:            self._release('거리 확보')
    elif a.nav_status in (NAV_SUCCEEDED, NAV_ABORTED, NAV_CANCELED):
                                                    self._release('leader 주행 종료')
    elif waited > self._resume_timeout:            self._release('대기 시간 초과')

② 멈출 것이 있을 때만

③ 시계가 튀면 무시

class LinkWatch:
    def __init__(self, timeout, restore_grace=1.0): ...

    def on_command(self, now, heartbeat=False):
        """명령을 받았다. 무장/복구 전이가 일어나면 그 이름을 돌려준다."""
        self.last_command = now
        if heartbeat and not self.armed and self.enabled:
            self.armed = True; self._last_poll = now
            return ARMED
        if self.lost:
            self.lost = False; self.restored_at = now
            return RESTORED

    def poll(self, now):
        """주기적으로 호출. 연결이 **방금** 끊어졌으면 LOST (전이 시점에 한 번만)."""
        last_poll, self._last_poll = self._last_poll, now
        if not self.armed or self.lost:
            return None
        # 시계가 튀었다 — 10 Hz 호출자에서 poll 간격 > timeout 은 있을 수 없다
        if last_poll is None or now < last_poll or (now - last_poll) > self.timeout:
            self.last_command = now
            return None
        if (now - self.last_command) < self.timeout:
            return None
        self.lost = True
        return LOST

    def alive(self, now):
        if not self.armed or not self.enabled:
            return True            # 판단 근거가 없으면 살아 있는 것으로 친다
        return not self.lost and (self.silence(now) or 0.0) < self.timeout

왜 카메라만으로는 안 되는가

분기가 있는 도로망에서 차선 추종은 세 가지를 모른다.

갈림길에서 어느 쪽
좌/우 차선 쌍이 둘 다 보인다
여기가 목적지인지
차선은 끝나지 않는다
교차로에서 어느 차선 쌍이 내 길인지
가로지르는 차선이 화면에 들어온다

셋 다 "맵 위 내 위치"가 있어야 풀린다. 라이다 맵과 mp1 의 AMCL 이 이미 있으므로 위치 파악용으로만 살린다.

반대로 AMCL 은 5 cm 쯤 흔들려 차선 중앙 유지에는 못 쓴다. 그건 카메라만 할 수 있다.

맵 경로 = 어느 길로  ·  카메라 = 그 길의 어디에
경로 (관제)
Dijkstra 로 엣지 열을 만들고 0.10 m 간격 waypoint 로 준다. 중심선 정확도는 ±5 cm 면 충분 — 나머지는 카메라가 메운다
조향 (로봇)
ω = ω_route(pure pursuit) + w_cam · ω_cam(카메라 PD). 카메라가 차선을 못 봐도 맵 경로가 끌고 간다
분기 (관제→로봇)
분기 노드 반경 0.25 m 안이면 JUNCTION — 카메라 보정을 끄고 맵 경로만. 교차로 차선 모호성을 풀지 않고 회피한다
상대 핑키 인식
카메라로 하지 않는다. 라이다(12.5 cm 높이)가 핑키(14 cm)를 잡고, 어차피 둘 다 "일단 멈춤"

def shortest_route(self, src, dst, step=DEFAULT_STEP):
    dist, prev, heap = {src: 0.0}, {}, [(0.0, src)]
    while heap:                                   # Dijkstra, 엣지 길이 = 비용
        d, node = heapq.heappop(heap)
        if node == dst: break
        for eid, other, forward in self.neighbors(node):   # oneway 는 한 방향만
            nd = d + self.edges[eid].length
            if nd < dist.get(other, float('inf')):
                dist[other] = nd; prev[other] = (node, eid, forward)
                heapq.heappush(heap, (nd, other))
    chain = [] ; node = dst
    while node != src:
        p, eid, forward = prev[node]; chain.append((eid, forward)); node = p
    chain.reverse()
    return self.build_route(chain, src, step)

def clear_until(self, name):
    slot = self.robots[name]
    if not slot.held: return 0                  # 아무것도 못 잡았으면 출발점 대기
    k = slot.held[-1]
    if k == slot.n_edges - 1: return slot.route.goal_idx    # 마지막 엣지면 목적지
    stop_s = max(slot.edge_start_s(k), slot.edge_end_s(k) - self.node_stop_margin)
    return slot.idx_at(stop_s)

def update(self, x, y):
    """pose 를 경로에 투영해 진행도를 갱신한다. 뒤로 튀는 투영은 무시."""
    lo = bisect.bisect_left(self.cum, self.progress_s - self.search_window)
    hi = bisect.bisect_right(self.cum, self.progress_s + self.search_window)
    best = None
    for i in range(max(0, lo - 1), min(len(self.points) - 1, hi)):
        (ax, ay), (bx, by) = self.points[i], self.points[i + 1]
        t, qx, qy = _project_segment(x, y, ax, ay, bx, by)
        d = math.hypot(x - qx, y - qy)
        if best is None or d < best[0]:
            seg = self.cum[i + 1] - self.cum[i]
            cross = (bx - ax) * (y - ay) - (by - ay) * (x - ax)
            best = (d, self.cum[i] + seg * t, cross / seg if seg > 1e-9 else 0.0)
    d, s, lateral = best
    if s >= self.progress_s - BACKTRACK_TOL:
        self.progress_s = max(self.progress_s, s)
        self.progress_idx = max(0, bisect.bisect_right(self.cum, self.progress_s + 1e-9) - 1)
    self.lateral, self.off_route = lateral, d
    return self.progress_idx

def steer(self, x, y, yaw, v, limit_idx=None, omega_max=1.5):
    """pure pursuit. lookahead 점이 뒤에 있으면 제자리 회전 방향만 준다."""
    lx, ly = self.lookahead_point(limit_idx)
    dx, dy = lx - x, ly - y
    c, s = math.cos(-yaw), math.sin(-yaw)
    rx, ry = dx * c - dy * s, dx * s + dy * c      # 로봇 좌표
    dist = math.hypot(rx, ry)
    alpha = math.atan2(ry, rx)
    if rx < 1e-6 or abs(alpha) >= math.pi / 2 - 1e-6:
        return math.copysign(omega_max, alpha), True   # 옆·뒤 → 회전만
    omega = 2.0 * max(v, 0.05) * math.sin(alpha) / max(dist, 1e-3)
    return max(-omega_max, min(omega_max, omega)), False

def command(self, omega_route, error_x, quality, dt, dist_to_stop, speed_factor=1.0,
            turn_in_place=False):
    """(v, ω). dist_to_stop 은 정지 지점까지 남은 호길이 (None 이면 제한 없음)."""
    p = self.p
    w_cam = CAM_WEIGHT.get(int(quality), 0.0) if error_x is not None else 0.0
    e = float(error_x or 0.0)
    de = 0.0 if self._prev_error is None else (e - self._prev_error) / dt
    self._prev_error = e if w_cam > 0 else None
    omega_cam = -(p.kp * e + p.kd * de) * p.cam_sign
    if turn_in_place:
        w_cam = 0.0          # 제자리 회전 중엔 기하가 무너져 카메라 항이 뜻이 없다
    omega = float(omega_route) + w_cam * omega_cam
    omega = max(-p.omega_max, min(p.omega_max, omega))

    if turn_in_place:
        v_target = 0.0
    else:
        v_target = p.v_max * min(1.0,
                                 max(0.0, 1.0 - p.k_err * abs(e) * w_cam),
                                 max(0.2, 1.0 - p.k_curv * abs(omega) / p.omega_max))
        v_target *= max(0.0, float(speed_factor))
        if dist_to_stop is not None:
            d = max(0.0, float(dist_to_stop))
            if d <= p.stop_tolerance:
                v_target = 0.0
            else:
                v_target = min(v_target, max(p.v_min_moving, (2.0 * p.a_decel * d) ** 0.5))
    if v_target > self._v_cmd:                       # 가속 슬루 (감속은 즉시)
        v_target = min(v_target, self._v_cmd + p.accel_slew * dt)
    self._v_cmd = v_target
    if v_target <= 0.0 and not turn_in_place:
        omega = 0.0
    return v_target, omega

        outer = _lane_points(lateral, heading, y_off + tape_width / 2.0)
        poly = [cam.project(X, Y) for X, Y in inner] + [... reversed(outer)]
        cv2.fillPoly(img, [np.array(poly, dtype=np.int32)], tape_bgr)
